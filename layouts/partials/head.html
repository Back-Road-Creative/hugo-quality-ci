{{/* Document head: title, description, canonical, Open Graph, Twitter card, feed autodiscovery, and the JSON-LD partial. Everything is driven by site params so the file is copied unchanged into any site. Nothing here assumes a CSS framework or a JS bundle -- list your own in params.stylesheets / params.scripts. Optional params: params.description fallback meta description params.tagline appended to the site title on the home page params.title_separator default "|" params.color_scheme "dark" or "light"; sets params.ogImage sharing image, path or absolute URL params.twitter_handle "@example" params.favicon e.g. "/favicon.svg" params.apple_touch_icon e.g. "/apple-touch-icon.png" params.webmanifest e.g. "/site.webmanifest" params.theme_color e.g. "#101010" params.stylesheets list of hrefs, emitted in order params.scripts list of src values, emitted with `defer` */ -}} {{- $p := .Site.Params -}} {{- $separator := $p.title_separator | default "|" -}} {{- with $p.color_scheme }} {{- /* Tell the user agent, so scrollbars and form controls match the page. */}} {{- end }} {{- /* Title: the home page gets "Site — tagline"; every other page gets "Page | Site", which keeps the site name in search results without burying the page name. */ -}} {{- $title := printf "%s %s %s" .Title $separator .Site.Title -}} {{- if .IsHome -}} {{- $title = .Site.Title -}} {{- with $p.tagline }}{{ $title = printf "%s — %s" $.Site.Title . }}{{ end -}} {{- end -}} {{- /* Description precedence: page front matter, then summary, then default. */ -}} {{- $description := $p.description -}} {{- if .Description -}} {{- $description = .Description -}} {{- else if and .IsPage .Summary -}} {{- $description = .Summary | plainify | chomp -}} {{- end -}} {{- $ogTitle := cond .IsHome $title .Title -}} {{- $image := "" -}} {{- with (.Params.ogImage | default $p.ogImage) -}}{{- $image = . | absURL -}}{{- end -}} {{- $imageAlt := .Site.Title -}} {{- with $p.tagline }}{{ $imageAlt = printf "%s — %s" $.Site.Title . }}{{ end -}} {{ $title }} {{- /* Taxonomy and term pages are thin by construction; keep them out of the index but let crawlers follow through to the real pages. */}} {{- if or (eq .Kind "term") (eq .Kind "taxonomy") }} {{- end }} {{- with $image }} {{- end }} {{- with $p.twitter_handle }} {{- end }} {{- with $image }} {{- end }} {{- /* Feed autodiscovery from the page's own output formats, so it stays correct when a section defines a narrower feed. */}} {{- with .OutputFormats.Get "rss" }} {{- end }} {{ partial "jsonld.html" . }} {{- with $p.favicon }} {{- end }} {{- with $p.apple_touch_icon }} {{- end }} {{- with $p.webmanifest }} {{- end }} {{- with $p.theme_color }} {{- end }} {{- range $p.stylesheets }} {{- end }} {{- range $p.scripts }} {{- end }}
